# Native macOS arm64 profile for the portable build (preset PortableMac, option GTASA_PORTABLE). Dependencies are built with Apple clang (Xcode) + the system libc++; the game itself is compiled by Homebrew LLVM clang (preset PortableMac, see CMakePresets.json).
# Use tools/portable/conan_install_mac.sh: it runs `conan install` from a scratch copy of conanfile.py so that the
# repo's ConanPresets.json (used by the Windows presets) and source/libs/imgui are never rewritten.
[settings]
os=Macos
arch=armv8
build_type=Release
compiler=apple-clang
compiler.cppstd=23
compiler.libcxx=libc++
compiler.version=21

[conf]
tools.cmake.cmaketoolchain:generator=Ninja
